#nullable enable
namespace WebUIToolkit.CommandLine
{
    public enum CommandExitCategory
    {
        Cancelled = 3,
        CommandFailure = 5,
        HostFailure = 6,
        Success = 0,
        Unavailable = 4,
        Usage = 1,
        Validation = 2,
    }
    public enum CommandOutputMode
    {
        Human = 0,
        Json = 1,
    }
    public enum CommandOutputModeSource
    {
        Default = 0,
        Environment = 1,
        ExplicitArgument = 2,
    }
    public interface IExitCodePolicy
    {
        int GetExitCode(WebUIToolkit.CommandLine.CommandExitCategory category);
    }
    public readonly record struct CommandOutputClassification
    {
        public bool IsValid { get; }
        public string? InvalidEnvironmentValue { get; }
        public WebUIToolkit.CommandLine.CommandOutputMode? Mode { get; }
        public WebUIToolkit.CommandLine.CommandOutputModeSource? Source { get; }
    }
    public sealed class CommandOutcome<T>
    {
        public T? Value { get; }
        public WebUIToolkit.CommandLine.CommandExitCategory ExitCategory { get; }
        public WebUIToolkit.CommandLine.CommandFault? Fault { get; }
        public bool IsSuccess { get; }
    }
    public sealed record CommandFault
    {
        public CommandFault(string code, string message, System.Collections.Generic.IReadOnlyDictionary<string, string>? details = null, bool retryable = false);
        public System.Collections.Generic.IReadOnlyDictionary<string, string> Details { get; }
        public bool Retryable { get; }
        public string Code { get; }
        public string Message { get; }
    }
    public sealed class DefaultExitCodePolicy : WebUIToolkit.CommandLine.IExitCodePolicy
    {
        public static WebUIToolkit.CommandLine.DefaultExitCodePolicy Instance { get; }
        public int GetExitCode(WebUIToolkit.CommandLine.CommandExitCategory category);
    }
    public static class CliProtocol
    {
        public const string Identity = "webuitoolkit.cli/1";
    }
    public static class CommandExitCodes
    {
        public const int Cancelled = 4;
        public const int CommandFailure = 10;
        public const int HostFailure = 70;
        public const int Success = 0;
        public const int Unavailable = 5;
        public const int Usage = 2;
        public const int Validation = 3;
        public static int GetDefault(WebUIToolkit.CommandLine.CommandExitCategory category);
    }
    public static class CommandOutputClassifier
    {
        public const string EnvironmentVariableName = "WEBUITOOLKIT_CLI_OUTPUT";
        public static WebUIToolkit.CommandLine.CommandOutputClassification Classify(WebUIToolkit.CommandLine.CommandOutputMode? explicitMode, string? environmentValue, WebUIToolkit.CommandLine.CommandOutputMode defaultMode = WebUIToolkit.CommandLine.CommandOutputMode.Human);
    }
    public static class CommandOutcome
    {
        public static WebUIToolkit.CommandLine.CommandOutcome<T> Failure<T>(WebUIToolkit.CommandLine.CommandExitCategory category, WebUIToolkit.CommandLine.CommandFault fault);
        public static WebUIToolkit.CommandLine.CommandOutcome<T> Success<T>(T value);
    }
}
